    <script>
        // Mapeamento dos textos exatos para cada tela baseados no design
        const labels = {
            'foco': 'FOCO',
            'shonen': 'SHONEM',
            'neon': 'NEON',
            'fdt': 'FIM DE\nTARDE',
            'corrida': 'CORRIDA',
            'sextap': 'SEX TAPE',
            'acordar': 'ACORDAR',
            'sozinho': 'SOZINHO',
            'party': 'NEON' // A foto 2 do Party curiosamente diz NEON embaixo.
        };

        // Função para alterar o tema e animar o layout
        function changeMood(mood) {
            // Atualiza o atributo no Body (que engatilha todo o CSS)
            document.body.setAttribute('data-mood', mood);
            
            // Atualiza o Label do texto na tela
            document.getElementById('label-mood').innerText = labels[mood];

            // Trata o texto "NOME DA MUSICA" pois alguns temas (FDT, Foco) tem layout linear
            const titleEl = document.getElementById('titulo');
            if(mood === 'fdt' || mood === 'foco' || mood === 'neon') {
                titleEl.innerHTML = "NOME DA MUSICA";
            } else {
                titleEl.innerHTML = "NOME DA<br>MUSICA";
            }
        }
    </script>
